Chapter 1: Pointers & Recursion

#include <stdio.h>
int factorial(int n) {
    return (n <= 1) ? 1 : n * factorial(n - 1);
}
int main() {
    printf("Factorial(6) = %d\n", factorial(6));
    return 0;
}
